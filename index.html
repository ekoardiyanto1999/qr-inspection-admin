import { createClient } from "https://esm.sh/@supabase/supabase-js@2"; const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "POST, OPTIONS", }; const json = (status: number, body: Record) => new Response(JSON.stringify(body), { status, headers: { ...corsHeaders, "Content-Type": "application/json" }, }); Deno.serve(async (req) => { if (req.method === "OPTIONS") { return new Response("ok", { headers: corsHeaders }); } if (req.method !== "POST") { return json(405, { error: "Method not allowed" }); } const supabaseUrl = Deno.env.get("SUPABASE_URL"); const anonKey = Deno.env.get("SUPABASE_ANON_KEY"); const serviceRoleKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY"); const appUrl = Deno.env.get("INSPECTOR_APP_URL"); if (!supabaseUrl || !anonKey || !serviceRoleKey) { return json(500, { error: "Server environment is not configured" }); } const authHeader = req.headers.get("Authorization") ?? ""; const token = authHeader.startsWith("Bearer ") ? authHeader.slice(7) : ""; if (!token) { return json(401, { error: "Missing bearer token" }); } const userClient = createClient(supabaseUrl, anonKey, { global: { headers: { Authorization: `Bearer ${token}` } }, auth: { persistSession: false, autoRefreshToken: false }, }); const adminClient = createClient(supabaseUrl, serviceRoleKey, { auth: { persistSession: false, autoRefreshToken: false }, }); // Validasi identitas pemanggil const { data: authData, error: authError } = await userClient.auth.getUser(); if (authError || !authData.user) { return json(401, { error: "Invalid or expired session" }); } const caller = authData.user; let body: Record; try { body = await req.json(); } catch { return json(400, { error: "Invalid JSON body" }); } const action = String(body.action ?? ""); // Pengguna hanya boleh mengganti password akunnya sendiri. if (action === "complete_password_setup") { const newPassword = String(body.new_password ?? ""); if (newPassword.length < 12) { return json(400, { error: "Password must contain at least 12 characters", }); } const currentMetadata = caller.app_metadata ?? {}; const { error } = await adminClient.auth.admin.updateUserById( caller.id, { password: newPassword, app_metadata: { ...currentMetadata, must_change_password: false, }, }, ); if (error) { return json(400, { error: error.message }); } return json(200, { success: true, message: "Password updated. Refresh the session to continue.", }); } // Semua tindakan pengelolaan akun hanya untuk Admin aktif. const { data: adminMembership, error: membershipError } = await adminClient .from("elodie_inspection_members") .select("organization_id, role, is_active") .eq("user_id", caller.id) .eq("is_active", true) .maybeSingle(); if ( membershipError || !adminMembership || adminMembership.role !== "admin" ) { return json(403, { error: "Only an active organization Admin can manage users", }); } if (action === "force_password_reset") { const targetUserId = String(body.target_user_id ?? "").trim(); if (!targetUserId) { return json(400, { error: "Target user ID is required" }); } const { data: targetMembership, error: targetMembershipError } = await adminClient .from("elodie_inspection_members") .select("user_id, role, is_active") .eq("organization_id", adminMembership.organization_id) .eq("user_id", targetUserId) .eq("is_active", true) .maybeSingle(); if ( targetMembershipError || !targetMembership || !["inspector", "supervisor"].includes(targetMembership.role) ) { return json(404, { error: "Active Inspector/Supervisor membership was not found in this organization", }); } const { data: targetAuth, error: targetAuthError } = await adminClient.auth.admin.getUserById(targetUserId); if (targetAuthError || !targetAuth.user) { return json(404, { error: "Target Auth user was not found" }); } const targetMetadata = targetAuth.user.app_metadata ?? {}; const { error: updateError } = await adminClient.auth.admin.updateUserById(targetUserId, { app_metadata: { ...targetMetadata, must_change_password: true, }, }); if (updateError) { return json(400, { error: updateError.message }); } return json(200, { success: true, method: action, user_id: targetUserId, must_change_password: true, message: "User will be required to change password at next login", }); } if ( action !== "create_manual" && action !== "invite_email" ) { return json(400, { error: "Unsupported action" }); } const fullName = String(body.full_name ?? "").trim(); const email = String(body.email ?? "").trim().toLowerCase(); const role = String(body.role ?? "").toLowerCase(); if ( !fullName || !email || !["admin", "supervisor", "inspector"].includes(role) ) { return json(400, { error: "Name, valid email, and role are required", }); } if ( fullName.length > 120 || email.length > 254 || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email) ) { return json(400, { error: "Invalid name or email" }); } const organizationId = adminMembership.organization_id; const redirectTo = appUrl || "https://ekoardiyanto1999.github.io/qr-inspection-mobile/"; // Hanya Inspector dan Supervisor yang wajib mengganti password. const mustChangePassword = role === "inspector" || role === "supervisor"; let createdUserId: string | null = null; if (action === "create_manual") { const initialPassword = String(body.initial_password ?? ""); if (initialPassword.length < 12) { return json(400, { error: "Initial password must contain at least 12 characters", }); } const { data, error } = await adminClient.auth.admin.createUser({ email, password: initialPassword, email_confirm: true, user_metadata: { full_name: fullName }, app_metadata: { must_change_password: mustChangePassword, }, }); if (error || !data.user) { return json(400, { error: error?.message ?? "Could not create user", }); } createdUserId = data.user.id; } else { const { data, error } = await adminClient.auth.admin.inviteUserByEmail(email, { redirectTo, data: { full_name: fullName }, }); if (error || !data.user) { return json(400, { error: error?.message ?? "Could not send invitation", }); } createdUserId = data.user.id; // Tetapkan flag wajib ganti password pada akun undangan. const currentMetadata = data.user.app_metadata ?? {}; const { error: metadataError } = await adminClient.auth.admin.updateUserById( createdUserId, { app_metadata: { ...currentMetadata, must_change_password: mustChangePassword, }, }, ); if (metadataError) { await adminClient.auth.admin.deleteUser(createdUserId); return json(400, { error: "Invitation account was rolled back because password setup status could not be configured: " + metadataError.message, }); } } // Buat keanggotaan organisasi. const { error: insertError } = await adminClient .from("elodie_inspection_members") .insert({ organization_id: organizationId, user_id: createdUserId, full_name: fullName, role, is_active: true, }); if (insertError) { await adminClient.auth.admin.deleteUser(createdUserId); return json(400, { error: "Account was rolled back because membership creation failed: " + insertError.message, }); } return json(200, { success: true, method: action, user_id: createdUserId, role, must_change_password: mustChangePassword, message: action === "invite_email" ? "Invitation sent and organization membership created" : mustChangePassword ? "Account created. User must change the initial password at first login" : "Admin account created", }); });
